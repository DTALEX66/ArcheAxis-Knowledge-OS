### 修复后代码

```typescript
async function importFile(file: File) {
  setImportReceipt({name:file.name,bytes:file.size,state:"正在检查原件"});
  if (file.size > 64 * 1024 * 1024) {
    setMessage("当前导入上限为 64 MiB，请选择不超过上限的原件。");
    setFailure(true);
    setImportReceipt({name:file.name,bytes:file.size,state:"未导入：超过大小上限"});
    return;
  }
  setImporting(true); setFailure(false); setMessage("正在导入原件…");
  try {
    const content = new Uint8Array(await file.arrayBuffer());
    let binary = "";
    for (let offset = 0; offset < content.length; offset += 8192)
      binary += String.fromCharCode(...content.subarray(offset, offset + 8192));

    // ✅ 修复：验证回执
    const imported = await coreCommand("source_import", {
      body: { name: file.name, content_base64: btoa(binary) }
    });

    // ✅ 修复：验证 source_id 和 sha256（参考 FolderIngest.tsx:123-124）
    if (typeof imported.source_id !== "string" || !imported.source_id
        || typeof imported.sha256 !== "string" || !/^[a-f0-9]{64}$/.test(imported.sha256)) {
      setImportReceipt({name:file.name,bytes:file.size,state:"导入回执未确认，请保留原件"});
      setMessage("导入回执未确认 source_id/sha256；请保留原件并重试。");
      setFailure(true);
      setImporting(false);
      return;
    }

    const list = await coreCommand<{ sources: SourceDto[] }>("sources_list");
    setSources(list.sources);
    setImportReceipt({name:file.name,bytes:file.size,state:"原件已保留；转换尚未执行"});
    setMessage("原件已导入；选择它开始阅读。");
  } catch {
    setImportReceipt({name:file.name,bytes:file.size,state:"导入未确认，请保留原件"});
    setMessage("导入未完成，请保留原件并重试。");
    setFailure(true);
  }
  finally { setImporting(false); }
}
```

### 修复说明

1. **捕获返回值**：保存 `source_import` 的返回值到 `imported` 变量
2. **验证 source_id**：检查为非空字符串
3. **验证 sha256**：检查为64位十六进制字符串（正则 `/^[a-f0-9]{64}$/`）
4. **失败处理**：回执无效时明确提示并保留 IMPORT_FAILED 状态
5. **提前返回**：验证失败时不继续后续操作

### 参考实现

此修复参考了 `FolderIngest.tsx` 第123-124行的正确模式。

```typescript
function handleRestoreRequest(saved: MaterialRequest) {
  const frozen = {
    ...saved.request,
    question: saved.request.question,
    purpose: saved.request.purpose,
    // ✅ 修复：恢复 selection/range
    selection: saved.request.selection,
  };
  setQuestion(saved.request.question);
  setPurpose(saved.request.purpose);
  // ✅ 修复：恢复 range 状态
  if (saved.request.selection) {
    setRange({
      start: saved.request.selection.start,
      end: saved.request.selection.end
    });
  } else {
    setRange({start: 0, end: 0});
  }

  executeMaterialQuestion(frozen);
}
```

**修复后代码**：
```typescript
// 参考 ConversationPanel.tsx:20 的正确模式
async function cancel() {
  try {
    await coreCommand("material_question_cancel", { request_id });
    setMessage("取消请求已发送");
    setCancelling(false);
  } catch (error) {
    // ✅ 修复：添加错误处理
    setMessage("取消请求发送失败；原请求状态未知，请查询确认。");
    setFailure(true);
    setFailureReason(coreFailureReason(error));
    setCancelling(false);
  }
}

// 按钮调用保持不变
<button onClick={() => void cancel()}>取消</button>
```

3. **网络断开**：显示错误，不进入假的"已取消"状态
